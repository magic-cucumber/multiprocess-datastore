cmake_minimum_required(VERSION 3.18)
project(shared_counter_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(counter-helper src/counter_helper.cpp)
get_filename_component(TEST_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
target_sources(counter-helper PRIVATE "${TEST_REPO_ROOT}/coreTest/common/src/nonAndroidTest/native/helper_logger.cpp")
target_include_directories(counter-helper PRIVATE "${TEST_REPO_ROOT}/platform/common/native/include"
    "${TEST_REPO_ROOT}/coreTest/common/src/nonAndroidTest/native")
set_target_properties(counter-helper PROPERTIES MACOSX_BUNDLE FALSE)
target_compile_features(counter-helper PRIVATE cxx_std_17)
target_link_libraries(counter-helper PRIVATE Threads::Threads)
if(WIN32)
    target_compile_definitions(counter-helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    target_compile_options(counter-helper PRIVATE /EHsc /W4 /utf-8)
else()
    target_compile_options(counter-helper PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
    target_link_options(counter-helper PRIVATE -municode)
endif()
set_target_properties(counter-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(counter-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
