cmake_minimum_required(VERSION 3.18)
project(file_observer_tests LANGUAGES CXX)
add_executable(observer-helper src/observer_helper.cpp)
set_target_properties(observer-helper PROPERTIES MACOSX_BUNDLE FALSE)
target_compile_features(observer-helper PRIVATE cxx_std_17)
if(WIN32)
    target_compile_definitions(observer-helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    target_compile_options(observer-helper PRIVATE /EHsc /W4 /utf-8)
else()
    target_compile_options(observer-helper PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
    target_link_options(observer-helper PRIVATE -municode)
endif()
set_target_properties(observer-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(observer-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
