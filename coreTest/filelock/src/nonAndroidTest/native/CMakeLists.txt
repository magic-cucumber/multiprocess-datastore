cmake_minimum_required(VERSION 3.18)
project(file_lock_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(lock-helper src/lock_helper.cpp)
get_filename_component(TEST_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
target_sources(lock-helper PRIVATE "${TEST_REPO_ROOT}/coreTest/common/src/nonAndroidTest/native/helper_logger.cpp")
target_include_directories(lock-helper PRIVATE "${TEST_REPO_ROOT}/platform/common/native/include"
    "${TEST_REPO_ROOT}/coreTest/common/src/nonAndroidTest/native")
set_target_properties(lock-helper PROPERTIES MACOSX_BUNDLE FALSE)
target_compile_features(lock-helper PRIVATE cxx_std_17)
target_link_libraries(lock-helper PRIVATE Threads::Threads)
if(MINGW)
    target_link_options(lock-helper PRIVATE -municode)
endif()
set_target_properties(lock-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(lock-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
